cmake_minimum_required(VERSION 3.16)
project(rustdesk_entry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RUSTDESK_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../../libs/arm64-v8a/liblibrustdesk.so")
if(NOT EXISTS "${RUSTDESK_CORE}")
  message(FATAL_ERROR "Build and stage the real ARM64 Rust core before building HAP")
endif()
add_library(rustdesk_core SHARED IMPORTED)
set_target_properties(rustdesk_core PROPERTIES IMPORTED_LOCATION "${RUSTDESK_CORE}")
add_library(entry SHARED napi_bridge.cpp renderer.cpp)
set_target_properties(entry PROPERTIES SKIP_BUILD_RPATH TRUE)
# The SDK toolchain supplies an unused --gcc-toolchain flag to Clang.
target_compile_options(entry PRIVATE -Wall -Wextra -Werror -Wno-unused-command-line-argument)
target_link_options(entry PRIVATE -Wl,--no-undefined)
target_link_libraries(entry PRIVATE rustdesk_core libace_napi.z.so libace_ndk.z.so libEGL.so libGLESv2.so)
