cmake_minimum_required(VERSION 3.16)
project(rustdesk_capture_probe LANGUAGES CXX)
option(RD_BUILD_OHOS_CAPTURE_PROBE "Build the standalone development capture library" OFF)
if(RD_BUILD_OHOS_CAPTURE_PROBE)
  add_library(capture_probe SHARED frame.cpp configure.cpp)
  set_target_properties(capture_probe PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON SKIP_BUILD_RPATH TRUE)
  target_compile_options(capture_probe PRIVATE -Wall -Wextra -Werror -Wno-unused-command-line-argument)
  target_link_options(capture_probe PRIVATE -Wl,--no-undefined)
  target_link_libraries(capture_probe PRIVATE libnative_avscreen_capture.so)
endif()
